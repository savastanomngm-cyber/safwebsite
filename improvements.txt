Key structural decisions:

    Python is the brain; the browser is only an eye. All computation moves server-side. The web terminal becomes a read-only dashboard over a FastAPI endpoint. This kills the CORS-proxy problem, the XSS surface, and the key-exposure problem in one move.
    One shared core library. The current three Python files duplicate llm(), extract_json(), clear_yf_cache(), period_return() four times each. Divergent copies = divergent numbers. Single source of truth.
    SQLite instead of JSON files. Baskets, memory, AI audit log, and price cache in one transactional DB. Concurrent CLI + web access stops corrupting state.
    No analysis without data. The system physically cannot render a signal unless it comes from fetched prices + grounded inputs. SIM mode becomes sandbox mode — clearly watermarked, never producing tradeable-looking output.

A (Alpha Independence, 30) — continued. The full function:
python

def a_score(px, spy):
    """Alpha Independence — beta-adjusted residual IR, sign-gated.
    Uncorrelated LOSERS now score ~0 instead of 25."""
    common = px.index.intersection(spy.index)
    rt = px.adj_close[common].pct_change().dropna()
    rs = spy.adj_close[common].pct_change().dropna()
    if len(rt) < 120:
        return 0.0
    beta = rt.cov(rs) / rs.var()
    resid = rt - beta * rs                    # idiosyncratic return stream
    ir = (resid.mean() / resid.std()) * np.sqrt(252)   # residual information ratio
    corr = rt.corr(rs)
    # Gate: independence only pays when the residual actually made money
    gate = np.tanh(resid.sum() * 10)          # smooth sign: -1..+1
    raw = (1 - abs(corr)) * 15 + np.clip(ir, -2, 2) * 7.5
    return float(np.clip(gate * raw, -15, 30))

Note this sub-score can now go negative — an uncorrelated disaster actively subtracts from the composite. That's the whole point.

R (Relative Strength, 20) — replace the free +12 floor with a proper risk-adjusted spread:

def r_score(px, spy):
    common = px.index.intersection(spy.index)
    rt = px.adj_close[common].pct_change(63).iloc[-1]   # 3-month
    rs = spy.adj_close[common].pct_change(63).iloc[-1]
    vol = px.adj_close.pct_change().std() * np.sqrt(252)
    # Excess return per unit of own volatility, no free floor
    return float(np.clip((rt - rs) / max(vol * np.sqrt(0.25), 0.05) * 5, -10, 20))

def q_score(fund):
    """Quality proxy from REAL fundamentals (no synthFund)."""
    gm = fund.get("gross_margin") or 0
    om = fund.get("oper_margin") or 0
    roe = fund.get("returnOnEquity") or 0
    # Gross margin >40% = pricing power signal (kept from v3 thesis,
    # but now it's real data and only worth 15 pts, not implicit everywhere)
    s = 0.0
    s += np.clip((gm - 0.25) / 0.45, 0, 1) * 8      # margin level
    s += np.clip(om / max(gm, 0.01), 0, 1) * 4      # op/gross conversion efficiency
    s += np.clip(roe / 0.30, 0, 1) * 3              # capital efficiency
    return float(np.clip(s, 0, 15))

# saf/quant/backtest.py
def walk_forward_score_test(universe_tickers, start="2019-01-01"):
    """
    Monthly rebalance walk-forward:
      - At each month-end, compute Score v2 using ONLY data up to that date
      - Rank universe, form top-quintile vs bottom-quintile portfolios
      - Measure forward 1M returns, net of costs
    Output: quintile spread, IC series, turnover, decay curve
    """
    dates = month_ends(start)
    records = []
    for d in dates[:-1]:
        scores = {}
        for t in universe_tickers:
            px = load_prices_to(t, d)          # point-in-time slice — NO lookahead
            if len(px) < 250: continue
            fund = load_fundamentals_asof(t, d)  # cached snapshots, as-of dated
            scores[t] = shadow_alpha_v2(px, spy_slice(d), fund)["total"]
        df = pd.Series(scores).sort_values()
        q = np.array_split(df.index, 5)
        fwd = {t: fwd_return(t, d, months=1) for t in df.index}
        for i, bucket in enumerate(q):
            rets = [fwd[t] for t in bucket if t in fwd]
            records.append({"date": d, "quintile": i + 1,
                            "ret": np.mean(rets), "n": len(rets)})
    res = pd.DataFrame(records)
    return evaluate(res)

def evaluate(res):
    by_q = res.groupby("quintile").ret.mean() * 12   # annualized
    ic = res.groupby("date").apply(
        lambda g: g.sort_values("quintile").ret.corr(
                  range(1, 6), method="spearman"))
    return {
        "q5_minus_q1_annual": by_q[5] - by_q[1],
        "ic_mean": ic.mean(), "ic_ir": ic.mean() / ic.std(),
        "monotonic": bool(by_q.is_monotonic_increasing),
        "turnover_est": "~40%/mo typical for momentum-style",
    }

Decision rule written into the system: if q5_minus_q1_annual < 3% or ic_ir < 0.3 after costs, the score is demoted to "research indicator" status and the UI says so explicitly. The threshold stops being folklore.

def decay_curve(res, horizons=(5, 21, 63, 126)):

RUBRIC_GROUNDED_SYS = """You are a bottleneck analyst. You will receive an
EVIDENCE PACK containing excerpts from the company's own SEC filings.

Rules:
1. Score each criterion 1-5 ONLY if the evidence pack contains direct support.
2. For each score, quote the exact supporting sentence from the pack.
3. If no support exists, score the criterion 2 (neutral) and write
   "INSUFFICIENT EVIDENCE" as your quote.
4. Never use your own knowledge of the company to fill gaps.
5. Return ONLY valid JSON:
{"scores": {...}, "citations": {...}, "unsupported": [...], "total": N}

An unsupported high score is a FAILURE, not a guess."""

def grounded_rubric(ticker, pack):
    out = extract_json(llm(RUBRIC_GROUNDED_SYS,
        f"EVIDENCE PACK:\n{json.dumps(pack, indent=1)}",
        temperature=0.2, force_json=True))
    if not out: return None
    # Post-hoc verification: check citations actually appear in the pack
    for crit, quote in out.get("citations", {}).items():
        if quote != "INSUFFICIENT EVIDENCE" and quote not in pack_text(pack):
            out["scores"][crit] = 2          # hallucinated citation -> neutral score
            out.setdefault("flagged", []).append(crit)
    return out

The post-hoc citation verification is the enforcement mechanism — LLMs that fabricate quotes get their scores silently demoted to neutral. This converts hallucination from a silent poison into a measurable, flagged event.

    actions = []
    for pos in store.open_positions():
        px = load_prices(pos.ticker)
        last = px.close.iloc[-1]

        # 1. Stop check (hard rule, no AI involved)
        if last <= pos.stop:
            actions.append(close(pos, reason="STOP_HIT"))
            continue
        # 2. Time stop
        if days_since(pos.entry) > pos.time_stop_days and realized_ret(pos) < 0:
            actions.append(close(pos, reason="TIME_STOP"))
            continue
        # 3. Thesis decay — recompute score; exit if it collapsed
        s = shadow_alpha_v2(px, spy(), fund(pos.ticker))["total"]
        if s < 40:
            actions.append(trim(pos, fraction=0.5,
                          reason=f"SCORE_DECAY ({s:.0f})"))
        # 4. Trailing stop ratchet
        pos.stop = max(pos.stop, last - pos.trail_dist)
    store.commit()
    return actions

# saf/intraday/leadlag.py
import yfinance as yf
from statsmodels.tsa.stattools import coint, adfuller

def lead_lag_scan(equity, proxy, interval="5m", period="60d"):
    eq = yf.download(equity, interval=interval, period=period, prepost=False)
    px = yf.download(proxy,  interval=interval, period=period, prepost=False)
    # RTH alignment
    eq, px = align_rth(eq, px)
    re, rp = eq.close.pct_change().dropna(), px.close.pct_change().dropna()

    results = {}
    for lag in range(-5, 6):                      # negative = proxy leads
        if lag == 0:
            results[0] = re.corr(rp)
        elif lag > 0:
            results[lag] = re.iloc[:-lag].corr(rp.iloc[lag:])
        else:
            results[lag] = re.iloc[-lag:].corr(rp.iloc[:lag])

    best_lag = max(results, key=lambda k: abs(results[k]))
    spread = np.log(eq.close) - beta_of(eq, px) * np.log(px.close)

    return {
        "pair": f"{equity}->{proxy}",
        "best_lag_bars": best_lag,
        "corr_at_best": round(results[best_lag], 4),
        "min_overlap_ok": len(re) >= 30 * 78 // 5,   # ≥30 RTH days of 5m bars
        "spread_stationary": adfuller(spread.dropna())[1] < 0.05,
        "cointegrated": coint(eq.close, px.close)[1] < 0.05,
        # Multi-interval confirmation — actually computed now
        "confirmed_intervals": sum(
            lead_lag_scan_fast(equity, proxy, iv)["best_lag_bars"] == best_lag
            for iv in ("1m", "5m", "15m")),
        "tradeable": all([
            abs(results[best_lag]) >= 0.15,
            best_lag > 0,                              # equity must LEAD
            adfuller(spread.dropna())[1] < 0.05,       # mean-reverting spread
        ]),
    }

"""Config loading with validation. Fails loudly on bad config."""
import yaml
from pathlib import Path
from functools import lru_cache

ROOT = Path(__file__).resolve().parent.parent

class ConfigError(Exception): pass

@lru_cache(maxsize=1)
def load() -> dict:
    path = ROOT / "saf" / "universe.yaml"
    if not path.exists():
        raise ConfigError(f"Missing {path}")
    cfg = yaml.safe_load(path.read_text())

    _validate(cfg)
    return cfg

def _validate(cfg):
    s = cfg.get("settings", {})
    if "benchmark" not in s:
        raise ConfigError("settings.benchmark required")
    th = s.get("score_thresholds", {})
    if not (0 < th.get("watch", 45) < th.get("candidate", 60) <= 100):
        raise ConfigError("thresholds must satisfy 0 < watch < candidate <= 100")

    seen = set()
    for b in cfg.get("baskets", []):
        name = b.get("name")
        if not name or not b.get("holdings"):
            raise ConfigError(f"basket needs name+holdings: {b}")
        dupes = [t for t in b["holdings"] if t in seen]
        if dupes:
            print(f"⚠️  WARNING: {dupes} appear in multiple baskets "
                  f"(allowed, but check intent)")
        wsum = sum(b["holdings"].values())
        if not 0 < wsum < 100:
            raise ConfigError(f"basket '{name}' weight sum suspicious: {wsum}")
        seen.update(b["holdings"])

def baskets(cfg=None) -> dict:
    """name -> holdings dict"""
    cfg = cfg or load()
    return {b["name"]: b["holdings"] for b in cfg["baskets"]}

def all_tickers(cfg=None) -> list:
    cfg = cfg or load()
    ts = set()
    for b in cfg["baskets"]:
        ts.update(b["holdings"])
    for group in cfg.get("screening_universe", {}).values():
        ts.update(group)
    ts.add(cfg["settings"]["benchmark"])
    return sorted(ts)

def sector_of(ticker, cfg=None):
    cfg = cfg or load()
    for sector, tickers in cfg.get("screening_universe", {}).items():
        if ticker in tickers:
            return sector
    for b in cfg["baskets"]:
        if ticker in b["holdings"]:
            return b.get("section", "BASKET")
    return None

        entry = {"ts": r["ts"], "kind": r["kind"],
                 "payload": json.loads(r["payload"]), "prev": prev}
        expect = hashlib.sha256(
            json.dumps(entry, sort_keys=True).encode()).hexdigest()[:16]
        if expect != r["hash"]:
            return False
        prev = r["hash"]
    return True

        if px.empty: continue
        entry_date = pd.Timestamp(r["date"])
        future = px[px.index > entry_date]
        if len(future) < 2: continue
        ret = (future["px"].iloc[-1] / future["px"].iloc[0] - 1) * 100
        outcome = "WIN" if ret > win else "LOSS" if ret < loss else "FLAT"
        c.execute("UPDATE memory SET outcome=?, realized_ret=? WHERE id=?",
                  (outcome, round(ret, 2), r["id"]))
    c.commit()

"""Incremental fetcher with quality gates. Never re-downloads what it has."""
import yfinance as yf
import pandas as pd
import numpy as np
from . import config, store

def refresh_ticker(ticker: str, lookback_days=None) -> bool:
    """Fetch only missing dates. Returns True if data is usable."""
    lookback = lookback_days or config.load()["settings"]["lookback_days"]
    last = store.last_price_date(ticker)
    start = (pd.Timestamp(last) + pd.Timedelta(days=1)) if last \
            else pd.Timestamp.now() - pd.Timedelta(days=lookback)
    if pd.Timestamp(start).date() >= pd.Timestamp.now().date():
        return True   # already current

    try:
        raw = yf.download(ticker, start=start.strftime("%Y-%m-%d"),
                          auto_adjust=True, progress=False, threads=False)
        if isinstance(raw.columns, pd.MultiIndex):
            raw.columns = raw.columns.get_level_values(0)
        if raw.empty:
            _record_failure(ticker)
            return False
        raw = raw.rename(columns=str.lower)
        raw["adj_close"] = raw.get("adj_close", raw["close"])
        store.upsert_prices(ticker, raw.dropna(subset=["close"]))
        flags = compute_quality_flags(ticker)
        store.set_flags(ticker, flags)
        return True
    except Exception as e:
        _record_failure(ticker)
        store.audit_log("data_error", {"ticker": ticker, "err": str(e)[:120]})
        return False
